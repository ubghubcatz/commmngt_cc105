use commngtcc105;

CREATE TABLE g4_EmployeeDetails (
    EmployeeID NVARCHAR(50) PRIMARY KEY, 
    EmployeeName NVARCHAR(255),
    FullAddress NVARCHAR(255),
    Position NVARCHAR(100)
);

select * from g4_EmployeeDetails;


INSERT INTO g4_EmployeeDetails (EmployeeID, EmployeeName, FullAddress, Position)
SELECT 
    EmployeeID, 
    LTRIM(RTRIM(FirstName + ' ' + COALESCE(NULLIF(MiddleName, ''), '') + ' ' + LastName)) AS EmployeeName,
    COALESCE(HouseNumber, '') + ', ' + 
    COALESCE(StreetAddress, '') + ', ' + 
    COALESCE(Barangay, '') + ', ' + 
    COALESCE(City, '') + ', ' + 
    COALESCE(Province, '') AS FullAddress,
    Position
FROM g4_EmployeesInfo;

    AttendanceID INT IDENTITY(1,1) PRIMARY KEY,
    EmployeeID NVARCHAR(50) FOREIGN KEY REFERENCES g4_EmployeeDetails(EmployeeID), 
    ScanTime DATETIME DEFAULT GETDATE(),
    ScanType NVARCHAR(10) CHECK (ScanType IN ('IN', 'OUT'))
);

use commngtcc105;

CREATE TABLE g4_ShiftSchedule (
    ShiftID INT PRIMARY KEY IDENTITY (1,1),
    EmployeeID NVARCHAR(50),
	EmployeeName NVARCHAR(255),
    Role VARCHAR(50),
    ShiftDate DATE,
    StartTime TIME,
    EndTime TIME,
	CONSTRAINT FK_Shift_Employee FOREIGN KEY (EmployeeID)
	REFERENCES g4_EmployeeDetails(EmployeeID)
	ON DELETE CASCADE
);

select * from g4_ShiftSchedule;

UPDATE g4_ShiftSchedule
SET EmployeeName = (SELECT EmployeeName FROM g4_EmployeeDetails WHERE g4_EmployeeDetails.EmployeeID = g4_ShiftSchedule.EmployeeID);

ALTER TABLE g4_ShiftSchedule
ALTER COLUMN EmployeeName NVARCHAR(255) NOT NULL;
